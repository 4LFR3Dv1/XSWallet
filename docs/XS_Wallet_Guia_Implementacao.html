3.3 Preimage Criptografada

⚠️ CRÍTICO: A preimage R atualmente é salva em plaintext no DB. Isso viola o spec de segurança.

Implementação

Arquivo: core/internal/vault/vault.go

// Adicionar métodos ao Vault
func (v *Vault) EncryptPreimage(preimage []byte) ([]byte, error) {
    v.mu.RLock()
    defer v.mu.RUnlock()
    
    if !v.unlocked || v.seed == nil {
        return nil, ErrVaultLocked
    }
    
    // Derivar chave de criptografia do seed
    key := derivePreimageKey(v.seed)
    
    // Criptografar com AES-256-GCM
    nonce, ciphertext, err := encrypt(preimage, key[:])
    if err != nil {
        return nil, err
    }
    
    // Retornar nonce || ciphertext
    return append(nonce, ciphertext...), nil
}

func (v *Vault) DecryptPreimage(encrypted []byte) ([]byte, error) {
    v.mu.RLock()
    defer v.mu.RUnlock()
    
    if !v.unlocked || v.seed == nil {
        return nil, ErrVaultLocked
    }
    
    // Separar nonce e ciphertext
    nonce := encrypted[:nonceLength]
    ciphertext := encrypted[nonceLength:]
    
    key := derivePreimageKey(v.seed)
    return decrypt(ciphertext, nonce, key[:])
}

3.4 PIN Lockout Persistente

Implementação

Arquivo: core/internal/vault/lockout.go

package vault

import (
    "database/sql"
    "time"
    "github.com/xs-wallet/xscore/internal/db"
)

type LockoutManager struct {
    db          *db.DB
    maxAttempts int           // 10
    lockoutTime time.Duration // 30 min
}

func NewLockoutManager(database *db.DB) *LockoutManager {
    return &LockoutManager{
        db:          database,
        maxAttempts: 10,
        lockoutTime: 30 * time.Minute,
    }
}

func (l *LockoutManager) RecordAttempt(success bool) error {
    if success {
        // Reset attempts
        _, err := l.db.Exec(`
            UPDATE vault_lockout SET attempts = 0, locked_until = NULL
            WHERE id = 1
        `)
        return err
    }
    
    // Increment attempts
    _, err := l.db.Exec(`
        INSERT INTO vault_lockout (id, attempts, last_attempt_at)
        VALUES (1, 1, datetime('now'))
        ON CONFLICT(id) DO UPDATE SET
            attempts = attempts + 1,
            last_attempt_at = datetime('now'),
            locked_until = CASE
                WHEN attempts + 1 >= ? THEN datetime('now', '+30 minutes')
                ELSE locked_until
            END
    `, l.maxAttempts)
    return err
}

func (l *LockoutManager) CanAttempt() (bool, time.Duration, error) {
    var attempts int
    var lockedUntil sql.NullString
    
    err := l.db.QueryRow(`
        SELECT attempts, locked_until FROM vault_lockout WHERE id = 1
    `).Scan(&attempts, &lockedUntil)
    
    if err == sql.ErrNoRows {
        return true, 0, nil
    }
    if err != nil {
        return false, 0, err
    }
    
    if lockedUntil.Valid {
        until, _ := time.Parse("2006-01-02 15:04:05", lockedUntil.String)
        if time.Now().Before(until) {
            return false, time.Until(until), nil
        }
    }
    
    return true, 0, nil
}

Integração no Vault:

4.1 Submarine Auto-Lock

Problema

Atualmente CreateFromQuote() deixa o swap em estado open sem lockup_address. O usuário precisa chamar Lock() manualmente.

Solução

// core/internal/swap/submarine.go
func (so *SubmarineOrchestrator) CreateFromQuote(ctx, quoteID) (*Swap, error) {
    // Validar quote
    quote, err := so.quotes.GetQuote(ctx, quoteID)
    if err != nil {
        return nil, err
    }
    
    // Criar swap
    swap, err := so.engine.Create(ctx, KindSubmarine, "regtest", 0)
    if err != nil {
        return nil, err
    }
    
    // AUTO-LOCK imediato
    swap, err = so.Lock(ctx, swap.ID, quoteID)
    if err != nil {
        return nil, err
    }
    
    return swap, nil
}

package swap

import (
    "github.com/btcsuite/btcd/txscript"
    "github.com/btcsuite/btcd/btcec/v2"
)

// VerifyP2TRAddress rebuilds o endereço P2TR localmente e compara
func VerifyP2TRAddress(
    claimPubkey, refundPubkey []byte,
    preimageHash []byte,
    timeoutBlocks int64,
    expectedAddress string,
) error {
    // 1. Construir swap tree
    tree, err := BuildSwapTree(KindSubmarine, SwapTreeParams{
        ClaimPubkey:   claimPubkey,
        RefundPubkey:  refundPubkey,
        PreimageHash:  preimageHash,
        TimeoutBlocks: timeoutBlocks,
    })
    if err != nil {
        return err
    }
    
    // 2. Gerar endereço P2TR
    address, err := tree.ToAddress(&chaincfg.RegressionNetParams)
    if err != nil {
        return err
    }
    
    // 3. Comparar
    if address != expectedAddress {
        return fmt.Errorf("address mismatch: got %s, want %s", address, expectedAddress)
    }
    
    return nil
}

    InternalKey *btcec.PublicKey

4.5 MuSig2 Implementation

Arquivo: core/internal/musig2/session.go

package musig2

import (
    "github.com/btcsuite/btcd/btcec/v2"
    "github.com/btcsuite/btcd/btcec/v2/schnorr/musig2"
)

type Session struct {
    ID           string
    SwapID       string
    UserKey      *btcec.PublicKey
    ProviderKey  *btcec.PublicKey
    AggKey       *musig2.AggregateKey
    SecNonce     [musig2.SecNonceSize]byte  // CRÍTICO: nunca expor
    PubNonce     [musig2.PubNonceSize]byte
    AggNonce     *musig2.Nonces
    PartialSig   *musig2.PartialSignature
}

func NewSession(swapID string, userPriv *btcec.PrivateKey, providerPub *btcec.PublicKey) (*Session, error) {
    // Agregar chaves (ordem lexicográfica)
    keys := []*btcec.PublicKey{userPriv.PubKey(), providerPub}
    sort.Slice(keys, func(i, j int) bool {
        return bytes.Compare(keys[i].SerializeCompressed(), keys[j].SerializeCompressed()) < 0
    })
    
    aggKey, err := musig2.AggregateKeys(keys, false)
    if err != nil {
        return nil, err
    }
    
    return &Session{
        ID:          uuid.New().String(),
        SwapID:      swapID,
        UserKey:     userPriv.PubKey(),
        ProviderKey: providerPub,
        AggKey:      aggKey,
    }, nil
}

func (s *Session) GenerateNonce() (pubNonce []byte, err error) {
    // Gerar nonce com aux randomness
    var auxRand [32]byte
    rand.Read(auxRand[:])
    
    secNonce, pubNonce, err := musig2.GenNonces(
        musig2.WithCustomRand(bytes.NewReader(auxRand[:])),
    )
    if err != nil {
        return nil, err
    }
    
    s.SecNonce = *secNonce
    copy(s.PubNonce[:], pubNonce[:])
    
    return s.PubNonce[:], nil
}

func (s *Session) AggregateNonces(providerNonce []byte) error {
    nonces := [][musig2.PubNonceSize]byte{s.PubNonce}
    var provNonce [musig2.PubNonceSize]byte
    copy(provNonce[:], providerNonce)
    nonces = append(nonces, provNonce)
    
    aggNonce, err := musig2.AggregateNonces(nonces)
    if err != nil {
        return err
    }
    
    s.AggNonce = aggNonce
    return nil
}

func (s *Session) Sign(txHash []byte, userPriv *btcec.PrivateKey) (partialSig []byte, err error) {
    var msg [32]byte
    copy(msg[:], txHash)
    
    partialSig, err = musig2.Sign(
        &s.SecNonce,
        userPriv,
        s.AggNonce,
        s.AggKey.FinalKey,
        msg,
    )
    if err != nil {
        return nil, err
    }
    
    // Cleanup secnonce
    for i := range s.SecNonce {
        s.SecNonce[i] = 0
    }
    
    return partialSig, nil
}

func (s *Session) Cleanup() {
    // Zero sensitive data
    for i := range s.SecNonce {
        s.SecNonce[i] = 0
    }
}

// Adicionar handler para WS updates
func (w *Watcher) HandleWSUpdate(swapID, boltzStatus string, tx *boltz.TxInfo) {
    // 1. Buscar swap
    swap, err := w.engine.Get(context.Background(), swapID)
    if err != nil {
        log.Printf("Failed to get swap %s: %v", swapID, err)
        return
    }
    
    // 2. Normalizar status usando boltz.Normalize existente
    mapping := boltz.Normalize(boltzStatus, swap.Kind)
    
    // 3. Se requer ação, reconciliar
    if mapping.Action != boltz.ActionNone {
        go w.reconcileSwap(context.Background(), swap)
    }
}
