3.3 Preimage Criptografada

⚠️ CRÍTICO: A preimage R atualmente é salva em plaintext no DB. Isso viola o spec de segurança.

Implementação

Arquivo: core/internal/vault/vault.go

// Adicionar métodos ao Vault
func (v *Vault) EncryptPreimage(preimage []byte) ([]byte, error) {
    v.mu.RLock()
    defer v.mu.RUnlock()
    
    if !v.unlocked || v.seed == nil {
        return nil, ErrVaultLocked
    }
    
    // Derivar chave de criptografia do seed
    key := derivePreimageKey(v.seed)
    
    // Criptografar com AES-256-GCM
    nonce, ciphertext, err := encrypt(preimage, key[:])
    if err != nil {
        return nil, err
    }
    
    // Retornar nonce || ciphertext
    return append(nonce, ciphertext...), nil
}

func (v *Vault) DecryptPreimage(encrypted []byte) ([]byte, error) {
    v.mu.RLock()
    defer v.mu.RUnlock()
    
    if !v.unlocked || v.seed == nil {
        return nil, ErrVaultLocked
    }
    
    // Separar nonce e ciphertext
    nonce := encrypted[:nonceLength]
    ciphertext := encrypted[nonceLength:]
    
    key := derivePreimageKey(v.seed)
    return decrypt(ciphertext, nonce, key[:])
}

3.4 PIN Lockout Persistente

Implementação

Arquivo: core/internal/vault/lockout.go

package vault

import (
    "database/sql"
    "time"
    "github.com/xs-wallet/xscore/internal/db"
)

type LockoutManager struct {
    db          *db.DB
    maxAttempts int           // 10
    lockoutTime time.Duration // 30 min
}

func NewLockoutManager(database *db.DB) *LockoutManager {
    return &LockoutManager{
        db:          database,
        maxAttempts: 10,
        lockoutTime: 30 * time.Minute,
    }
}

func (l *LockoutManager) RecordAttempt(success bool) error {
    if success {
        // Reset attempts
        _, err := l.db.Exec(`
            UPDATE vault_lockout SET attempts = 0, locked_until = NULL
            WHERE id = 1
        `)
        return err
    }
    
    // Increment attempts
    _, err := l.db.Exec(`
        INSERT INTO vault_lockout (id, attempts, last_attempt_at)
        VALUES (1, 1, datetime('now'))
        ON CONFLICT(id) DO UPDATE SET
            attempts = attempts + 1,
            last_attempt_at = datetime('now'),
            locked_until = CASE
                WHEN attempts + 1 >= ? THEN datetime('now', '+30 minutes')
                ELSE locked_until
            END
    `, l.maxAttempts)
    return err
}

func (l *LockoutManager) CanAttempt() (bool, time.Duration, error) {
    var attempts int
    var lockedUntil sql.NullString
    
    err := l.db.QueryRow(`
        SELECT attempts, locked_until FROM vault_lockout WHERE id = 1
    `).Scan(&attempts, &lockedUntil)
    
    if err == sql.ErrNoRows {
        return true, 0, nil
    }
    if err != nil {
        return false, 0, err
    }
    
    if lockedUntil.Valid {
        until, _ := time.Parse("2006-01-02 15:04:05", lockedUntil.String)
        if time.Now().Before(until) {
            return false, time.Until(until), nil
        }
    }
    
    return true, 0, nil
}

package lnd

import (
    "context"
    "google.golang.org/grpc"
    "github.com/lightningnetwork/lnd/lnrpc"
)

type Client struct {
    conn      *grpc.ClientConn
    lightning lnrpc.LightningClient
}

func NewClient(address string, tlsCert, macaroon []byte) (*Client, error) {
    // Setup TLS + macaroon credentials
    creds := credentials.NewTLS(&tls.Config{
        Certificates: []tls.Certificate{cert},
    })
    
    conn, err := grpc.Dial(address, grpc.WithTransportCredentials(creds))
    if err != nil {
        return nil, err
    }
    
    return &Client{
        conn:      conn,
        lightning: lnrpc.NewLightningClient(conn),
    }, nil
}

// Mínimo viável
func (c *Client) GetInfo(ctx context.Context) (*lnrpc.GetInfoResponse, error) {
    return c.lightning.GetInfo(ctx, &lnrpc.GetInfoRequest{})
}

func (c *Client) PayInvoice(ctx context.Context, bolt11 string) (*lnrpc.SendResponse, error) {
    return c.lightning.SendPaymentSync(ctx, &lnrpc.SendRequest{
        PaymentRequest: bolt11,
    })
}

func (c *Client) AddInvoice(ctx context.Context, amount int64, memo string) (*lnrpc.AddInvoiceResponse, error) {
    return c.lightning.AddInvoice(ctx, &lnrpc.Invoice{
        Value: amount,
        Memo:  memo,
    })
}

// Stream de invoices
func (c *Client) SubscribeInvoices(ctx context.Context) (<-chan *lnrpc.Invoice, error) {
    stream, err := c.lightning.SubscribeInvoices(ctx, &lnrpc.InvoiceSubscription{})
    if err != nil {
        return nil, err
    }
    
    invoices := make(chan *lnrpc.Invoice)
    go func() {
        defer close(invoices)
        for {
            inv, err := stream.Recv()
            if err != nil {
                return
            }
            invoices <- inv
        }
    }()
    
    return invoices, nil
}

func (c *Client) LookupInvoice(ctx context.Context, paymentHash string) (*lnrpc.Invoice, error) {
    hash, _ := hex.DecodeString(paymentHash)
    return c.lightning.LookupInvoice(ctx, &lnrpc.PaymentHash{
        RHash: hash,
    })
}

package bitcoin

import (
    "bytes"
    "encoding/json"
    "net/http"
)

type Client struct {
    rpcURL   string
    user     string
    password string
    client   *http.Client
}

func NewClient(rpcURL, user, password string) *Client {
    return &Client{
        rpcURL:   rpcURL,
        user:     user,
        password: password,
        client:   &http.Client{Timeout: 30 * time.Second},
    }
}

type RPCRequest struct {
    JSONRPC string        `json:"jsonrpc"`
    Method  string        `json:"method"`
    Params  []interface{} `json:"params"`
    ID      int           `json:"id"`
}

func (c *Client) call(method string, params ...interface{}) (json.RawMessage, error) {
    req := RPCRequest{
        JSONRPC: "1.0",
        Method:  method,
        Params:  params,
        ID:      1,
    }
    
    body, _ := json.Marshal(req)
    httpReq, _ := http.NewRequest("POST", c.rpcURL, bytes.NewReader(body))
    httpReq.SetBasicAuth(c.user, c.password)
    httpReq.Header.Set("Content-Type", "application/json")
    
    resp, err := c.client.Do(httpReq)
    if err != nil {
        return nil, err
    }
    defer resp.Body.Close()
    
    var rpcResp struct {
        Result json.RawMessage `json:"result"`
        Error  *struct {
            Message string `json:"message"`
        } `json:"error"`
    }
    
    json.NewDecoder(resp.Body).Decode(&rpcResp)
    if rpcResp.Error != nil {
        return nil, fmt.Errorf("RPC error: %s", rpcResp.Error.Message)
    }
    
    return rpcResp.Result, nil
}

// Mínimo viável
func (c *Client) GetBlockHeight(ctx context.Context) (int64, error) {
    result, err := c.call("getblockcount")
    if err != nil {
        return 0, err
    }
    
    var height int64
    json.Unmarshal(result, &height)
    return height, nil
}

func (c *Client) GetTransaction(ctx context.Context, txid string) (*Transaction, error) {
    result, err := c.call("getrawtransaction", txid, true)
    if err != nil {
        return nil, err
    }
    
    var tx Transaction
    json.Unmarshal(result, &tx)
    return &tx, nil
}

func (c *Client) BroadcastTransaction(ctx context.Context, hex string) (string, error) {
    result, err := c.call("sendrawtransaction", hex)
    if err != nil {
        return "", err
    }
    
    var txid string
    json.Unmarshal(result, &txid)
    return txid, nil
}

func (c *Client) GetConfirmations(ctx context.Context, txid string) (int, error) {
    tx, err := c.GetTransaction(ctx, txid)
    if err != nil {
        return 0, err
    }
    return tx.Confirmations, nil
}

package watcher

import (
    "context"
    "time"
    "github.com/xs-wallet/xscore/internal/adapters/bitcoin"
    "github.com/xs-wallet/xscore/internal/adapters/liquid"
    "github.com/xs-wallet/xscore/internal/swap"
)

type ChainWatcher struct {
    btc    *bitcoin.Client
    liquid *liquid.Client
    engine *swap.Engine
}

func NewChainWatcher(btc *bitcoin.Client, liquid *liquid.Client, engine *swap.Engine) *ChainWatcher {
    return &ChainWatcher{
        btc:    btc,
        liquid: liquid,
        engine: engine,
    }
}

func (w *ChainWatcher) WatchTransaction(ctx context.Context, chain, txid string) error {
    ticker := time.NewTicker(30 * time.Second)
    defer ticker.Stop()
    
    for {
        select {
        case <-ctx.Done():
            return ctx.Err()
        case <-ticker.C:
            var confirmations int
            var err error
            
            if chain == "BTC" {
                confirmations, err = w.btc.GetConfirmations(ctx, txid)
            } else {
                confirmations, err = w.liquid.GetConfirmations(ctx, txid)
            }
            
            if err != nil {
                continue
            }
            
            if confirmations >= 1 {
                // Trigger reconcile
                go w.reconcileByTxid(ctx, txid)
                return nil
            }
        }
    }
}

func (w *ChainWatcher) GetBlockHeight(ctx context.Context, chain string) (int64, error) {
    if chain == "BTC" {
        return w.btc.GetBlockHeight(ctx)
    }
    return w.liquid.GetBlockHeight(ctx)
}

5.1 Submarine Auto-Lock

Problema

Atualmente CreateFromQuote() deixa o swap em estado open sem lockup_address. O usuário precisa chamar Lock() manualmente.

Solução

// core/internal/swap/submarine.go
func (so *SubmarineOrchestrator) CreateFromQuote(ctx, quoteID) (*Swap, error) {
    // Validar quote
    quote, err := so.quotes.GetQuote(ctx, quoteID)
    if err != nil {
        return nil, err
    }
    
    // Criar swap
    swap, err := so.engine.Create(ctx, KindSubmarine, "regtest", 0)
    if err != nil {
        return nil, err
    }
    
    // AUTO-LOCK imediato
    swap, err = so.Lock(ctx, swap.ID, quoteID)
    if err != nil {
        return nil, err
    }
    
    return swap, nil
}

    hash := sha256.Sum256(preimage)
    
    // 2. Criar swap
    swap, err := ro.engine.Create(ctx, KindReverse, "regtest", 0)
    if err != nil {
        return nil, err
    }
    
    // 3. Criptografar e salvar preimage
    encPreimage, err := ro.vault.EncryptPreimage(preimage)
    if err != nil {
        return nil, err
    }
    
    // 4. Atualizar DB
    _, err = ro.db.ExecContext(ctx, `
        UPDATE swaps SET encrypted_preimage = ?, preimage_hash_hex = ?
        WHERE id = ?
    `, encPreimage, hex.EncodeToString(hash[:]), swap.ID)
    
    // 5. Lock com provider
    return ro.Lock(ctx, swap.ID, quoteID)
}

// VerifyP2TRAddress rebuilds o endereço P2TR localmente e compara
func VerifyP2TRAddress(
    claimPubkey, refundPubkey []byte,
    preimageHash []byte,
    timeoutBlocks int64,
    expectedAddress string,
) error {
    // 1. Construir swap tree
    tree, err := BuildSwapTree(KindSubmarine, SwapTreeParams{
        ClaimPubkey:   claimPubkey,
        RefundPubkey:  refundPubkey,
        PreimageHash:  preimageHash,
        TimeoutBlocks: timeoutBlocks,
    })
    if err != nil {
        return err
    }
    
    // 2. Gerar endereço P2TR
    address, err := tree.ToAddress(&chaincfg.RegressionNetParams)
    if err != nil {
        return err
    }
    
    // 3. Comparar
    if address != expectedAddress {
        return fmt.Errorf("address mismatch: got %s, want %s", address, expectedAddress)
    }
    
    return nil
}

5.5 MuSig2 Implementation

Arquivo: core/internal/musig2/session.go

package musig2

import (
    "github.com/btcsuite/btcd/btcec/v2"
    "github.com/btcsuite/btcd/btcec/v2/schnorr/musig2"
)

type Session struct {
    ID           string
    SwapID       string
    UserKey      *btcec.PublicKey
    ProviderKey  *btcec.PublicKey
    AggKey       *musig2.AggregateKey
    SecNonce     [musig2.SecNonceSize]byte  // CRÍTICO: nunca expor
    PubNonce     [musig2.PubNonceSize]byte
    AggNonce     *musig2.Nonces
    PartialSig   *musig2.PartialSignature
}

func NewSession(swapID string, userPriv *btcec.PrivateKey, providerPub *btcec.PublicKey) (*Session, error) {
    // Agregar chaves (ordem lexicográfica)
    keys := []*btcec.PublicKey{userPriv.PubKey(), providerPub}
    sort.Slice(keys, func(i, j int) bool {
        return bytes.Compare(keys[i].SerializeCompressed(), keys[j].SerializeCompressed()) < 0
    })
    
    aggKey, err := musig2.AggregateKeys(keys, false)
    if err != nil {
        return nil, err
    }
    
    return &Session{
        ID:          uuid.New().String(),
        SwapID:      swapID,
        UserKey:     userPriv.PubKey(),
        ProviderKey: providerPub,
        AggKey:      aggKey,
    }, nil
}

func (s *Session) GenerateNonce() (pubNonce []byte, err error) {
    // Gerar nonce com aux randomness
    var auxRand [32]byte
    rand.Read(auxRand[:])
    
    secNonce, pubNonce, err := musig2.GenNonces(
        musig2.WithCustomRand(bytes.NewReader(auxRand[:])),
    )
    if err != nil {
        return nil, err
    }
    
    s.SecNonce = *secNonce
    copy(s.PubNonce[:], pubNonce[:])
    
    return s.PubNonce[:], nil
}

func (s *Session) AggregateNonces(providerNonce []byte) error {
    nonces := [][musig2.PubNonceSize]byte{s.PubNonce}
    var provNonce [musig2.PubNonceSize]byte
    copy(provNonce[:], providerNonce)
    nonces = append(nonces, provNonce)
    
    aggNonce, err := musig2.AggregateNonces(nonces)
    if err != nil {
        return err
    }
    
    s.AggNonce = aggNonce
    return nil
}

func (s *Session) Sign(txHash []byte, userPriv *btcec.PrivateKey) (partialSig []byte, err error) {
    var msg [32]byte
    copy(msg[:], txHash)
    
    partialSig, err = musig2.Sign(
        &s.SecNonce,
        userPriv,
        s.AggNonce,
        s.AggKey.FinalKey,
        msg,
    )
    if err != nil {
        return nil, err
    }
    
    // Cleanup secnonce
    for i := range s.SecNonce {
        s.SecNonce[i] = 0
    }
    
    return partialSig, nil
}

func (s *Session) Cleanup() {
    // Zero sensitive data
    for i := range s.SecNonce {
        s.SecNonce[i] = 0
    }
}

package nodemanager

import (
    "context"
    "os/exec"
    "time"
)

type NodeManager struct {
    nodes    map[string]*ManagedNode
    dataDir  string
    network  string
}

type ManagedNode struct {
    Name      string
    Binary    string
    Version   string
    DataDir   string
    Process   *exec.Cmd
    Health    HealthChecker
    Status    NodeStatus
}

type NodeStatus string

const (
    StatusStopped  NodeStatus = "stopped"
    StatusStarting NodeStatus = "starting"
    StatusRunning  NodeStatus = "running"
    StatusError    NodeStatus = "error"
)

func NewNodeManager(dataDir, network string) *NodeManager {
    return &NodeManager{
        nodes:   make(map[string]*ManagedNode),
        dataDir: dataDir,
        network: network,
    }
}

func (nm *NodeManager) Start(ctx context.Context, nodeName string) error {
    node := nm.nodes[nodeName]
    if node == nil {
        return fmt.Errorf("node %s not registered", nodeName)
    }
    
    // 1. Verificar binário (hash check)
    if err := nm.verifyBinary(node); err != nil {
        return fmt.Errorf("binary verification failed: %w", err)
    }
    
    // 2. Spawn process
    node.Status = StatusStarting
    if err := nm.spawnProcess(ctx, node); err != nil {
        node.Status = StatusError
        return err
    }
    
    // 3. Aguardar health check
    if err := nm.waitForHealth(ctx, node, 30*time.Second); err != nil {
        nm.Stop(nodeName)
        return err
    }
    
    node.Status = StatusRunning
    return nil
}
